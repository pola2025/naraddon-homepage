Google Sheets 생성

새 스프레드시트를 만들고 첫 행에 헤더를 입력합니다.

시트 이름:

상담접수

헤더 (첫 행에 입력):

접수일시 이름 회사명 연락처 이메일 사업자번호 상담유형 연매출 직원수 희망시간 지역 요청내용

⚠️ 스프레드시트 공유 설정: "링크가 있는 모든 사용자" → "뷰어"로 설정

2

Google Apps Script 배포

  1. 스프레드시트에서 확장 프로그램 → Apps Script 클릭
  2. 기존 코드 전체 삭제 후 아래 코드 붙여넣기
  3. 배포 → 새 배포 클릭
  4. 유형: 웹 앱 선택
  5. 실행 주체: 나
  6. 액세스 권한: 모든 사용자
  7. 배포 후 웹 앱 URL 복사
// 파일 위치: docs/google-apps-script-sheets.js 참조
// 또는 아래 핵심 코드 사용

const SHEET_NAME = '상담접수';

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const data = payload.data;

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME);

    const newRow = [
      data.createdAt || new Date().toLocaleString('ko-KR'),
      data.name || '',
      data.company || '',
      data.phone || '',
      data.email || '',
      data.businessNumber || '',
      data.consultType || '',
      data.annualRevenue || '',
      data.employeeCount || '',
      data.preferredTime || '',
      data.region || '',
      data.message || ''
    ];

    sheet.appendRow(newRow);

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, message: '웹훅 활성화됨' }))
    .setMimeType(ContentService.MimeType.JSON);
}
